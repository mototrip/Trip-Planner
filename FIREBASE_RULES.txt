rules_version = '2';
service cloud.firestore {
 match /databases/{database}/documents {
  function loggedIn() {return request.auth != null;}
  function isMember(rideId) {
   return loggedIn() && exists(/databases/$(database)/documents/sharedRides/$(rideId)/members/$(request.auth.uid));
  }
  function goodGroup(d) {
   return d.keys().hasAll(['ownerUid','name','members','expenses','transfers','createdAt','updatedAt'])
    && d.keys().hasOnly(['ownerUid','name','members','expenses','transfers','createdAt','updatedAt'])
    && d.ownerUid is string && d.name is string && d.name.size() <= 100
    && d.members is list && d.members.size() <= 100
    && d.expenses is list && d.expenses.size() <= 3000
    && d.transfers is list && d.transfers.size() <= 3000
    && d.createdAt is string && d.updatedAt is string;
  }
  match /sharedRides/{rideId} {
   allow get: if isMember(rideId);
   allow list: if false;
   allow create: if loggedIn()
      && goodGroup(request.resource.data)
      && request.resource.data.ownerUid == request.auth.uid;
   allow update: if isMember(rideId)
      && goodGroup(request.resource.data)
      && request.resource.data.ownerUid == resource.data.ownerUid
      && request.resource.data.createdAt == resource.data.createdAt
      && request.resource.data.diff(resource.data).affectedKeys()
           .hasOnly(['name','members','expenses','transfers','updatedAt']);
   allow delete: if loggedIn() && resource.data.ownerUid == request.auth.uid;
   match /members/{userId} {
    allow get: if loggedIn() && (request.auth.uid == userId ||
     get(/databases/$(database)/documents/sharedRides/$(rideId)).data.ownerUid == request.auth.uid);
    allow list: if false;
    allow create: if loggedIn() && request.auth.uid == userId
       && request.resource.data.keys().hasAll(['name','inviteId','joinedAt'])
       && request.resource.data.keys().hasOnly(['name','inviteId','joinedAt'])
       && request.resource.data.name is string
       && request.resource.data.name.size() > 0 && request.resource.data.name.size() <= 80
       && request.resource.data.inviteId is string
       && request.resource.data.joinedAt is string
       && (get(/databases/$(database)/documents/sharedRides/$(rideId)).data.ownerUid == userId
         || (exists(/databases/$(database)/documents/sharedRideInvites/$(request.resource.data.inviteId))
            && get(/databases/$(database)/documents/sharedRideInvites/$(request.resource.data.inviteId)).data.rideId == rideId));
    allow update: if false;
    allow delete: if loggedIn() && (request.auth.uid == userId ||
     get(/databases/$(database)/documents/sharedRides/$(rideId)).data.ownerUid == request.auth.uid);
   }
  }
  match /sharedRideInvites/{inviteId} {
   allow get: if loggedIn();
   allow list: if false;
   allow create: if loggedIn()
    && request.resource.data.keys().hasAll(['ownerUid','rideId','createdAt'])
    && request.resource.data.keys().hasOnly(['ownerUid','rideId','createdAt'])
    && request.resource.data.ownerUid == request.auth.uid
    && request.resource.data.rideId is string
    && get(/databases/$(database)/documents/sharedRides/$(request.resource.data.rideId)).data.ownerUid == request.auth.uid;
   allow update: if false;
   allow delete: if loggedIn() && resource.data.ownerUid == request.auth.uid;
  }
  match /{document=**} { allow read, write: if false; }
 }
}
